SpawnRules = {}

SpawnRules.hazardIntervalMin = 5.5
SpawnRules.hazardIntervalMax = 8.0
SpawnRules.missileInitialDelayMin = 4.0
SpawnRules.missileInitialDelayMax = 7.0
SpawnRules.missileIntervalMin = 9.0
SpawnRules.missileIntervalMax = 13.0
SpawnRules.missileVolleyDelayMin = 0.55
SpawnRules.missileVolleyDelayMax = 0.9

function SpawnRules.GetEnemySpawnInterval(elapsedTime, startInterval, acceleration, minInterval, levelId)
    local value = startInterval - elapsedTime * acceleration
    if value < minInterval then
        value = minInterval
    end
    if levelId == 2 then
        value = value * 0.88
    end
    return value
end

local function Lerp(minimum, maximum, randomValue)
    local value = math.max(0, math.min(1, randomValue or 0))
    return minimum + (maximum - minimum) * value
end

function SpawnRules.GetHazardSpawnInterval(randomValue, levelId, configuredMin, configuredMax)
    local interval = Lerp(SpawnRules.hazardIntervalMin, SpawnRules.hazardIntervalMax, randomValue)
    return levelId == 2 and interval * 0.9 or interval
end

function SpawnRules.GetSpikeChance(levelId)
    return levelId == 2 and 0.65 or 0.58
end

function SpawnRules.GetMissileInitialDelay(randomValue, levelId)
    local delay = Lerp(SpawnRules.missileInitialDelayMin, SpawnRules.missileInitialDelayMax, randomValue)
    return levelId == 2 and delay * 0.85 or delay
end

function SpawnRules.GetMissileSpawnInterval(randomValue, levelId)
    local interval = Lerp(SpawnRules.missileIntervalMin, SpawnRules.missileIntervalMax, randomValue)
    return levelId == 2 and interval * 0.9 or interval
end

function SpawnRules.GetMissileVolleyDelay(randomValue, levelId)
    local delay = Lerp(SpawnRules.missileVolleyDelayMin, SpawnRules.missileVolleyDelayMax, randomValue)
    return levelId == 2 and delay * 0.9 or delay
end

function SpawnRules.GetMissileVolleyCount(randomValue, levelId)
    local value = randomValue or 0
    local singleThreshold = levelId == 2 and 0.42 or 0.5
    local doubleThreshold = levelId == 2 and 0.76 or 0.82
    if value < singleThreshold then
        return 1
    end
    if value < doubleThreshold then
        return 2
    end
    return 3
end

function SpawnRules.GetSpikeGroupCount(randomValue, levelId)
    local value = randomValue or 0
    local singleThreshold = levelId == 2 and 0.42 or 0.5
    local doubleThreshold = levelId == 2 and 0.78 or 0.82
    if value < singleThreshold then
        return 1
    end
    if value < doubleThreshold then
        return 2
    end
    return 3
end
