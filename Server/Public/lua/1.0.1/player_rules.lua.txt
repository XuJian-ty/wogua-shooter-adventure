PlayerRules = {}

PlayerRules.postShieldInvincibleDuration = 3.0
PlayerRules.growthWarningDuration = 3.0
PlayerRules.magnetForwardRange = 3.0

function PlayerRules.GetJumpForce(jumpNumber, firstJumpForce, secondJumpForce)
    if jumpNumber == 0 then
        return firstJumpForce
    end
    return secondJumpForce
end

function PlayerRules.GetGlideHoldDelay(defaultValue)
    return defaultValue
end

function PlayerRules.GetGlideDescentSpeed(defaultValue)
    return defaultValue
end

function PlayerRules.ShouldBlink(damageInvincibleTime, growthTime)
    return damageInvincibleTime > 0 or
        (growthTime > 0 and growthTime <= PlayerRules.growthWarningDuration)
end
